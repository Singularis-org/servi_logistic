  ```sql
  SELECT o."id", o."nombre", o."cobro_plataforma_por_orden" AS valor_propio,
         p."nombre" AS plan, p."costo_por_orden" AS plan_fijo, p."costo_por_orden_pct" AS plan_pct,
         p."costo_por_orden_minimo" AS plan_minimo,
         (SELECT t."monto" FROM "cobro_plataforma_tarifa" t) AS valor_global

   WHERE o."cobro_plataforma_por_orden" IS NOT NULL
   ORDER BY o."nombre";
  ```
